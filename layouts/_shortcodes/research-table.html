{{- /* Page-local plain-text data keeps the online table aligned with its source. */ -}} {{- $id := .Get "id" -}} {{- $resource := .Page.Resources.GetMatch "assets/web-tables.json" -}} {{- if not $resource -}} {{- errorf "research-table in %s requires assets/web-tables.json" .Page.File.Path -}} {{- else -}} {{- $tables := $resource.Content | transform.Unmarshal -}} {{- $table := index $tables $id -}} {{- if not $table -}} {{- errorf "Unknown research-table %q in %s" $id .Page.File.Path -}} {{- else -}} {{- $uid := printf "research-table-%s-%d" ($id | urlize) .Ordinal -}} {{- $caption := $table.caption | default $table.title -}} {{- $width := "standard" -}} {{- if eq $table.width "wide" -}}{{- $width = "wide" -}}{{- end -}} {{- $columnCount := len $table.header -}} {{- if or (not $table.title) (eq $columnCount 0) -}} {{- errorf "research-table %q in %s needs a title and header" $id .Page.File.Path -}} {{- end -}} {{- end -}} {{- end -}}
